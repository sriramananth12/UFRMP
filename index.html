  <script>

    /* ---------------------------------------------------------------

       SUPABASE CONFIGURATION & INITIALIZATION

    --------------------------------------------------------------- */

    const SUPABASE_URL = 'https://nejpqyqmcbchivhtovny.supabase.co';

    const SUPABASE_ANON_KEY = 'eyJhbGciOiJIUzI1NiIsInR5cCI6IkpXVCJ9.eyJpc3MiOiJzdXBhYmFzZSIsInJlZiI6Im5lanBxeXFtY2JjaGl2aHRvdm55Iiwicm9sZSI6ImFub24iLCJpYXQiOjE3OTA4MzY5OTUsImV4cCI6MjEwNjQxMjk5NX0.Dp3DpvDiLRY9PpOuKGDmrR5pH6mnRa3MTSH7NmGfNg0';

 

    const supabaseClient = window.supabase ? window.supabase.createClient(SUPABASE_URL, SUPABASE_ANON_KEY) : null;

 

    // Local DB state cache

    const DB = {

      works: [],

      ganttData: [],

      budgetMatrix: []

    };

 

    /* ---------------------------------------------------------------

       UTILITY FUNCTIONS

    --------------------------------------------------------------- */

    function fmtDate(dateStr) {

      if (!dateStr) return '';

      return new Date(dateStr).toLocaleDateString();

    }

 

    function clamp(val, min, max) {

      return Math.min(Math.max(val, min), max);

    }

 

    function esc(s) {

      if (s === undefined || s === null) return '';

      return String(s).replace(/[&<>"']/g, c => ({

        '&': '&amp;',

        '<': '&lt;',

        '>': '&gt;',

        '"': '&quot;',

        "'": '&#39;'

      }[c]));

    }

 

    function showStorageBanner(message) {

      const banner = document.getElementById('storageBanner');

      const text = document.getElementById('bannerText');

      if (banner && text) {

        text.textContent = message || 'Connected to Supabase live dataset.';

        banner.classList.remove('hidden');

      }

    }

 

    function showFileProtocolBanner() {

      if (window.location.protocol === 'file:') {

        showStorageBanner('Warning: Running directly via file:// protocol. Some browser features may be restricted.');

      }

    }

 

    /* ---------------------------------------------------------------

       SUPABASE DATA FETCHERS

    --------------------------------------------------------------- */

    async function loadWorks() {

      if (!supabaseClient) return [];

      const { data, error } = await supabaseClient.from('works').select('*');

      if (error) {

        console.warn('Could not load works table:', error.message);

        return [];

      }

      return data;

    }

 

    async function loadGantt() {

      if (!supabaseClient) return [];

      const { data, error } = await supabaseClient.from('gantt_data').select('*');

      if (error) {

        console.warn('Could not load gantt_data table:', error.message);

        return [];

      }

      return data;

    }

 

    async function loadBudgetMatrix() {

      if (!supabaseClient) return [];

      const { data, error } = await supabaseClient.from('budget_matrix').select('*');

      if (error) {

        console.warn('Could not load budget_matrix table:', error.message);

        return [];

      }

      return data;

    }

 

    /* ---------------------------------------------------------------

       UI RENDERER

    --------------------------------------------------------------- */

    function render() {

      document.getElementById('worksStatus').innerText = `${DB.works.length} records loaded`;

      document.getElementById('ganttStatus').innerText = `${DB.ganttData.length} tasks loaded`;

      document.getElementById('budgetStatus').innerText = `${DB.budgetMatrix.length} entries loaded`;

    }

 

    /* ---------------------------------------------------------------

       APPLICATION INITIALIZATION

    --------------------------------------------------------------- */

    async function initApp() {

      showFileProtocolBanner();

 

      try {

        const [works, gantt, budget] = await Promise.all([

          loadWorks(),

          loadGantt(),

          loadBudgetMatrix()

        ]);

 

        // Assign to active state

        DB.works = works || [];

        DB.ganttData = gantt || [];

        DB.budgetMatrix = budget || [];

 

        showStorageBanner('Successfully connected and synchronized with Supabase.');

      } catch (err) {

        console.error('Failed to initialize application data from Supabase:', err);

        showStorageBanner('Failed to load live data from Supabase.');

      }

 

      // Enable actions once hydrated

      const xlsxBtn = document.getElementById('xlsxFab');

      if (xlsxBtn) xlsxBtn.disabled = false;

 

      render();

    }

 

    // Execution entry point

    if (document.readyState === 'loading') {

      document.addEventListener('DOMContentLoaded', initApp);

    } else {

      initApp();

    }
